#define _perkmod_twin_spitfire_included

int g_iPerkTS;

void TwinSpitfire_Register()
{
	g_iPerkTS = Perks_Register("twin_spitfire", TREE_SPITTER);
	Events_Subscribe(PME_ABILITY, TS_Ability);
}

void TS_Ability(int client, int extra)
{
	if (!Perk_HasInfected(client, g_iPerkTS)) return;
	if (!Util_AbilityNameMatches(client, "ability_spit")) return;
	// второй плевок через задержку (pending-механизм, без читов)
	float delay = PerkCvar_Float(g_iPerkTS, "_delay", 0.5);
	DataPack dp;
	CreateDataTimer(delay, TS_Respit, dp, TIMER_FLAG_NO_MAPCHANGE);
	dp.WriteCell(GetClientUserId(client));
}

Action TS_Respit(Handle timer, DataPack dp)
{
	dp.Reset();
	int client = GetClientOfUserId(dp.ReadCell());
	if (!client || !IsValidClient(client)) return Plugin_Stop;
	if (GetClientTeam(client) != 3 || GetZombieClass(client) != 4) return Plugin_Stop;
	if (!Perk_HasInfected(client, g_iPerkTS)) return Plugin_Stop;
	// живая сверка: спавн кислоты в точке — натива left4dhooks / энтити-путь (README п.1)
	int proj = CreateEntityByName("spitter_projectile");
	if (proj > 0)
	{
		float pos[3];
		GetClientAbsOrigin(client, pos);
		pos[2] += 60.0;
		DispatchSpawn(proj);
		TeleportEntity(proj, pos, NULL_VECTOR, NULL_VECTOR);
	}
	return Plugin_Stop;
}
