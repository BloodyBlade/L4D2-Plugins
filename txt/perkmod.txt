public void OnPluginStart()
{
	Events_Init();
	Core_RegisterConvars();
	Core_RegisterAllPerks();
	Events_HookAll();
	Events_InitDmgSource();
	Events_Timers();
	Menu_RegisterCommands();
	CreateConVar("perkmod_version", PLUGIN_VERSION, "PerkMod rewrite version", FCVAR_NOTIFY|FCVAR_DONTRECORD);
}

public void OnConfigsExecuted()
{
	DB_Connect();
	Core_GiveBotPerks();
}

public void OnMapStart()
{
	Events_FireMapStart();
	HH_OnMapStart();
	Sp_OnMapStart();
}

public void OnClientPutInServer(int client)
{
	if (client > 0)
	{
		SDKHook(client, SDKHook_OnTakeDamage, Dmg_TakeDamage);
		CreateTimer(3.0, Tim_AutoShow, GetClientUserId(client), TIMER_FLAG_NO_MAPCHANGE); // авто-показ меню, 3 сек
	}
}

Action Tim_AutoShow(Handle timer, int userid)
{
	int c = GetClientOfUserId(userid);
	if (c > 0 && IsValidClient(c) && !g_bConfirmed[c] && !IsFakeClient(c)) Menu_ShowMain(c);
	return Plugin_Stop;
}

public void OnClientDisconnect(int client)
{
	if (client > 0)
	{
		HH_ResetClient(client);
		Sp_ResetClient(client);
		g_iSlot1[client] = g_iSlot2[client] = g_iSlot3[client] = 0;
		g_iInfectedPerk[client] = 0;
		g_bConfirmed[client] = false;
		Util_ClearSpeed(client);
	}
}

public void OnMapEnd()
{
	for (int i = 0; i <= MaxClients; i++)
	{
		g_iUBApplied[i] = 0;
		g_flSPReadyAt[i] = 0.0;
	}
}
