 #endinput
#endif
#define _perkmod_mega_adhesive_included

int g_iPerkADH;

void MegaAdhesive_Register()
{
	g_iPerkADH = Perks_Register("mega_adhesive", TREE_SPITTER);
	CreateTimer(0.5, ADH_Tick, 0, TIMER_REPEAT|TIMER_FLAG_NO_MAPCHANGE);
}

Action ADH_Tick(Handle timer)
{
	if (!g_hCvarEnable.BoolValue) return Plugin_Continue;
	for (int c = 1; c <= MaxClients; c++)
	{
		if (!IsValidClient(c) || GetClientTeam(c) != 2) continue;
		if (!Perk_HasInfected(c, g_iPerkADH)) continue;   // owner = игрок с перком
		// живая сверка: проп нахождения в кислоте (m_isInAcid / аналог) — README п.2
		bool inacid = view_as<bool>(GetEntProp(c, Prop_Send, "m_isInAcid"));
		if (inacid && g_iSpeedOwner[c] != SPEED_ADH)
			Util_SaveBaseSpeed(c, c), Util_SetSpeed(c, 0.7);
		else if (!inacid && g_iSpeedOwner[c] == SPEED_ADH)
			Util_RestoreOwnerSpeed(c);
	}
	return Plugin_Continue;
}
