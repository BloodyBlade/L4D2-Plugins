int Perks_Register(const char[] name, int tree)
{
	strcopy(g_sPerkNames[g_iPerkCount], 32, name);
	g_iPerkTree[g_iPerkCount] = tree;
	int id = g_iPerkCount++;
	// привязка id к util-конварам скорости (один раз, при регистрации соотв. перка)
	if (tree == TREE_BOOMER && strcmp(name, "metabolic_boost") == 0) {} // не бывает; см. tank ниже
	if (strcmp(name, "metabolic_boost") == 0) g_iIdMB = id;
	if (strcmp(name, "speeding_bullet") == 0) g_iIdSB = id;
	if (strcmp(name, "speed_demon")     == 0) g_iIdSD = id;
	if (strcmp(name, "drag_and_drop")   == 0) g_iIdDD = id;
	if (strcmp(name, "motion_sickness") == 0) g_iIdMS = id;
	if (strcmp(name, "extreme_conditioning") == 0) g_iIdEC = id;
	if (strcmp(name, "spirit")       == 0) g_iIdSpirit       = id;
	if (strcmp(name, "helping_hand") == 0) g_iIdHelpingHand  = id;
	return id;
}

void Perk_SetSlot(int client, int slot, int id)
{
	if (slot == 1) g_iSlot1[client] = id;
	else if (slot == 2) g_iSlot2[client] = id;
	else if (slot == 3) g_iSlot3[client] = id;
	else g_iInfectedPerk[client] = id;
	DB_SaveClient(client);
}

bool Perk_HasSurvivor(int client, int id)
{
	return IsValidClient(client) && GetClientTeam(client) == 2
		&& (g_iSlot1[client] == id || g_iSlot2[client] == id || g_iSlot3[client] == id);
}

bool Perk_HasInfected(int client, int id)
{
	return IsValidClient(client) && GetClientTeam(client) == 3 && g_iInfectedPerk[client] == id;
}

// оба пути подтверждения — через один метод
void Core_ApplyConfirmed(int client)
{
	if (g_iInfectedPerk[client] > 0) Events_Fire(PME_APPLY_CONFIRM, client, g_iInfectedPerk[client]);
	if (g_iSlot1[client] > 0 || g_iSlot2[client] > 0 || g_iSlot3[client] > 0)
		Events_Fire(PME_APPLY_CONFIRM, client, 0);
	g_bConfirmed[client] = true;
	Menu_ShowSummary(client);
}

// бот-пулы: "1,2,3" — id перков, раздаемых ботам (парсер переписан чисто)
void Core_RegisterConvars()
{
	g_hCvarEnable = CreateConVar("perkmod_enable", "1", "PerkMod enabled", FCVAR_NOTIFY);
	g_hCvarBotSur1 = CreateConVar("perkmod_bot_sur1", "1,2,3,4", "Bot pool sur1", FCVAR_NOTIFY);
	g_hCvarBotSur2 = CreateConVar("perkmod_bot_sur2", "1,2,3,4", "Bot pool sur2", FCVAR_NOTIFY);
	g_hCvarBotSur3 = CreateConVar("perkmod_bot_sur3", "1,2", "Bot pool sur3", FCVAR_NOTIFY);
	g_hCvarBotInf  = CreateConVar("perkmod_bot_infected", "1,2,3,4,5,6,7,8", "Bot pool infected", FCVAR_NOTIFY);
	AutoExecConfig(true, "perkmod");
}

void Core_GiveBotPerks()
{
	int sur1[8], sur2[8], sur3[8], inf[8];
	int n1 = Core_ParsePool(g_hCvarBotSur1, sur1, 8);
	int n2 = Core_ParsePool(g_hCvarBotSur2, sur2, 8);
	int n3 = Core_ParsePool(g_hCvarBotSur3, sur3, 8);
	int ni = Core_ParsePool(g_hCvarBotInf, inf, 8);
	for (int i = 1; i <= MaxClients; i++)
	{
		if (!IsValidClient(i) || !IsFakeClient(i) || g_bConfirmed[i]) continue;
		switch (GetClientTeam(i))
		{
			case 2:
			{
				Perk_SetSlot(i, 1, sur1[GetURandomInt() % n1]);
				Perk_SetSlot(i, 2, sur2[GetURandomInt() % n2]);
				Perk_SetSlot(i, 3, sur3[GetURandomInt() % n3]);
				Core_ApplyConfirmed(i);
			}
			case 3:
			{
				Perk_SetSlot(i, 4, inf[GetURandomInt() % ni]);
				Core_ApplyConfirmed(i);
			}
		}
	}
}

int Core_ParsePool(ConVar h, int[] out, int max)
{
	char buf[128];
	h.GetString(buf, sizeof(buf));
	int count = 0, pos = 0, len = strlen(buf);
	while (pos < len && count < max)
	{
		int val = 0;
		bool has = false;
		while (pos < len && buf[pos] >= '0' && buf[pos] <= '9')
		{
			val = val * 10 + (buf[pos] - '0');
			pos++; has = true;
		}
		if (has && val > 0 && val < PERKMOD_MAXPERKS) out[count++] = val;
		while (pos < len && (buf[pos] == ',' || buf[pos] == ' ')) pos++;
	}
	return count > 0 ? count : 1;
}
