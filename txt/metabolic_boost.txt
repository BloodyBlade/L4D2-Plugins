#if defined _perkmod_metabolic_boost_included
 #endinput
#endif
#define _perkmod_metabolic_boost_included

int g_iPerkMB;

void MetabolicBoost_Register()
{
	g_iPerkMB = Perks_Register("metabolic_boost", TREE_TANK);
	Events_Subscribe(PME_SPAWNTANK, MB_Spawn);
	Events_Subscribe(PME_SPAWN,     MB_Apply);
}

void MB_Apply(int client, int extra)
{
	if (!Perk_HasInfected(client, g_iPerkMB)) return;
	if (GetZombieClass(client) != 8) return;
	if (g_iSpeedOwner[client] != SPEED_NONE && g_iSpeedOwner[client] != SPEED_MB) return;

	Util_SaveBaseSpeed(client, client);
	Util_SetSpeed(client, PerkCvar_Float(g_iPerkMB, "_mult", 1.4));
}

void MB_Spawn(int client, int extra) { MB_Apply(client, extra); }
