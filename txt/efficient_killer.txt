#if defined _perkmod_efficient_killer_included
 #endinput
#endif
#define _perkmod_efficient_killer_included

int g_iPerkEK;

void EfficientKiller_Register()
{
	g_iPerkEK = Perks_Register("efficient_killer", TREE_HUNTER);
	Dmg_Subscribe(EK_Dmg);
}

void EK_Dmg(int victim, int attacker, float &dmg, int &dmgtype, int weapon)
{
	if (dmg <= 0.0) return;
	if (!Perk_HasInfected(attacker, g_iPerkEK)) return;
	if (GetZombieClass(attacker) != 3) return;
	float mult = PerkCvar_Float(g_iPerkEK, "_dmg", 1.3);
	if (mult < 1.0) mult = 1.0;
	if (mult > 4.0) mult = 4.0;
	dmg *= mult;
}
