#if defined _perkmod_double_tap_included
 #endinput
#endif
#define _perkmod_double_tap_included

int g_iPerkDTap;

void DoubleTap_Register()
{
	g_iPerkDTap = Perks_Register("double_tap", TREE_SUR1);
	FireRate_Subscribe(DT_AttackRate);
	HookEvent("weapon_reload", DT_Reload, EventHookMode_Post);
}

bool DT_IsSemiAuto(const char[] cls)
{
	static const char semis[][] = {
		"weapon_pistol", "weapon_pistol_magnum",
		"weapon_pumpshotgun", "weapon_shotgun_spas", "weapon_autoshotgun",
		"weapon_sniper_military", "weapon_sniper_scout", "weapon_sniper_awp", "weapon_sniper_desert"
	};
	for (int i = 0; i < sizeof(semis); i++)
		if (strcmp(cls, semis[i]) == 0) return true;
	return false;
}

float DT_AttackRate(int client, int weapon)
{
	if (!Perk_HasSurvivor(client, g_iPerkDTap)) return 1.0;
	if (weapon <= 0) return 1.0;
	char cls[40];
	GetEntityClassname(weapon, cls, sizeof(cls));
	if (!DT_IsSemiAuto(cls)) return 1.0;

	float rate = PerkCvar_Float(g_iPerkDTap, "_rate", 0.6667);   // кламп 0.2..0.9
	if (rate < 0.2) rate = 0.2;
	if (rate > 0.9) rate = 0.9;
	if (rate <= 0.0) return 1.0;
	return 1.0 / rate;   // мост ждёт множитель интервала
}

void DT_Reload(Event e, const char[] n, bool db)
{
	int c = GetClientOfUserId(e.GetInt("userid"));
	if (!Perk_HasSurvivor(c, g_iPerkDTap)) return;
	int wep = GetEntPropEnt(c, Prop_Send, "m_hActiveWeapon");
	if (wep <= 0) return;
	char cls[40];
	GetEntityClassname(wep, cls, sizeof(cls));
	if (StrContains(cls, "shotgun") == -1) return;
	// живая сверка: перезарядка пистолетов/снайперок в оригинале идёт тем же SoH-конвейером
	float rate = PerkCvar_Float(g_iPerkDTap, "_rate_reload", 0.8);   // кламп 0.2..1.0
	if (rate < 0.2) rate = 0.2;
	if (rate > 1.0) rate = 1.0;
	Util_ScaleShotgunReload(wep, cls, rate);
}
