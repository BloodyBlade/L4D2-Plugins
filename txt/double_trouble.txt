// perkmod/double_trouble.inc — танк призывает дубли (tank №2)
#if defined _perkmod_double_trouble_included
 #endinput
#endif
#define _perkmod_double_trouble_included

int g_iPerkDT;

void DoubleTrouble_Register()
{
	g_iPerkDT = Perks_Register("double_trouble", TREE_TANK);
	Events_Subscribe(PME_SPAWNTANK, DT_Spawn);
	Events_Subscribe(PME_SPAWN,     DT_Spawn);
}

void DT_Spawn(int client, int extra)
{
	if (!Perk_HasInfected(client, g_iPerkDT)) return;
	if (GetZombieClass(client) != 8) return;
	int count = PerkCvar_Int(g_iPerkDT, "_count", 1);
	for (int i = 0; i < count; i++)
		Util_QueuePendingTank(client);   // pending-механизм танковой машины, без чит-команд
}
