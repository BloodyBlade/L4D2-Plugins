// создание шин: вызвать один раз из OnPluginStart
void Events_Init()
{
	for (int i = 0; i < PME_MAX; i++)
	{
		if (g_hBus[i] != null) delete g_hBus[i];   // защита от повторного вызова
		g_hBus[i] = new ArrayList();
	}
	if (g_hMapStart != null) delete g_hMapStart;
	g_hMapStart = new ArrayList();
	if (g_hDmg      != null) delete g_hDmg;
	g_hDmg = new ArrayList();
	if (g_hFireRate != null) delete g_hFireRate;
	g_hFireRate = new ArrayList();
}

void Events_Subscribe(int type, Function cb)
{
	g_hBus[type].Push(cb);
}

void Events_Fire(int type, int client, int extra)
{
	for (int i = 0; i < g_hBus[type].Length; i++)
	{
		Function cb = g_hBus[type].Get(i);
		Call_StartFunction(GetMyHandle(), cb);
		Call_PushCell(client);
		Call_PushCell(extra);
		Call_Finish();
	}
}

// облитость: единый источник правды (dead_wreckening/barf_bagged/motion_sickness не дублируют)
void Ev_NowIt(Event e, const char[] n, bool db)
{
	int victim = GetClientOfUserId(e.GetInt("userid"));
	int boomer = GetClientOfUserId(e.GetInt("attacker"));
	if (victim > 0 && victim <= MaxClients && !g_bSlimed[victim])
	{
		g_bSlimed[victim] = true;
		g_iSlimedBy[victim] = boomer;
		g_iSlimed++;
		g_iSlimerLast = boomer;
	}
}

// мост скорости атаки: сокращаем m_flNextPrimaryAttack после выстрела
void Ev_WeaponFire(Event e, const char[] n, bool db)
{
	if (!g_hCvarEnable.BoolValue) return;
	int c = GetClientOfUserId(e.GetInt("userid"));
	int wep = GetEntPropEnt(c, Prop_Send, "m_hActiveWeapon");
	if (wep <= 0) return;
	float best = 1.0;
	for (int i = 0; i < g_hFireRate.Length; i++)
	{
		Function f = g_hFireRate.Get(i);
		Call_StartFunction(GetMyHandle(), f);
		Call_PushCell(c);
		Call_PushCell(wep);
		int ret; Call_Finish(ret);
		float m = view_as<float>(ret);   // float-возврат идёт ячейкой, биты те же
		if (m < best) best = m;
	}
	if (best >= 1.0) return;
	float next = GetEntPropFloat(wep, Prop_Send, "m_flNextPrimaryAttack");
	float interval = next - GetGameTime();
	if (interval > 0.0)
		SetEntPropFloat(wep, Prop_Send, "m_flNextPrimaryAttack", GetGameTime() + interval * best);
}

// источник шины урона (дельта 15)
void Events_InitDmgSource()
{
	for (int i = 1; i <= MaxClients; i++)
		if (IsValidClient(i)) SDKHook(i, SDKHook_OnTakeDamage, Dmg_TakeDamage);
}

Action Dmg_TakeDamage(int victim, int &attacker, int &inflictor, float &damage,
	int &damagetype, int &weapon, float damageForce[3], float damagePosition[3])
{
	if (!g_hCvarEnable.BoolValue) return Plugin_Continue;
	float dmg = damage; int dt = damagetype;
	for (int i = 0; i < g_hDmg.Length; i++)
	{
		Function f = g_hDmg.Get(i);
		Call_StartFunction(GetMyHandle(), f);
		Call_PushCell(victim);
		Call_PushCell(attacker);
		Call_PushFloatRef(dmg);      // float &dmg
		Call_PushCellRef(dt);        // int &dmgtype
		Call_PushCell(weapon);
		Call_Finish();
	}

	if (dmg != damage || dt != damagetype)
	{
		damage = dmg; damagetype = dt;
		return Plugin_Changed;
	}
	return Plugin_Continue;
}

// сверка облитости с состоянием движка раз в секунду
void Events_Timers()
{
	CreateTimer(1.0, Tim_SlimeCheck, 0, TIMER_REPEAT|TIMER_FLAG_NO_MAPCHANGE);
	CreateTimer(1.0, Tim_Pending,    0, TIMER_REPEAT|TIMER_FLAG_NO_MAPCHANGE);
}

Action Tim_SlimeCheck(Handle timer)
{
	for (int i = 1; i <= MaxClients; i++)
	{
		if (!IsValidClient(i) || GetClientTeam(i) != 2) continue;
		bool it = view_as<bool>(GetEntProp(i, Prop_Send, "m_it")); // живая сверка пропа
		if (!it && g_bSlimed[i])
		{
			g_bSlimed[i] = false;
			g_iSlimedBy[i] = 0;
			if (--g_iSlimed < 0) g_iSlimed = 0;
		}
	}
	return Plugin_Continue;
}

Action Tim_Pending(Handle timer)
{
	if (!g_hCvarEnable.BoolValue) return Plugin_Continue;
	Util_PumpPending();
	return Plugin_Continue;
}
