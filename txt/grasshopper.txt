#define _perkmod_grasshopper_included

int g_iPerkGH;

void Grasshopper_Register()
{
	g_iPerkGH = Perks_Register("grasshopper", TREE_HUNTER);
	Events_Subscribe(PME_POUNCE, GH_Pounce);
}

void GH_Pounce(int hunter, int victim)
{
	if (!Perk_HasInfected(hunter, g_iPerkGH)) return;
	float mult = PerkCvar_Float(g_iPerkGH, "_leap", 1.3);   // кламп 1.0..3.0
	if (mult < 1.0) mult = 1.0;
	if (mult > 3.0) mult = 3.0;
	float vel[3];
	GetEntPropVector(hunter, Prop_Data, "m_vecVelocity", vel);
	vel[0] *= mult; vel[1] *= mult; vel[2] *= mult;
	TeleportEntity(hunter, NULL_VECTOR, NULL_VECTOR, vel);
}
