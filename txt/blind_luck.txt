// perkmod/blind_luck.inc — кулдаун рвоты режется + жертва слепнет (boomer №2, метод Doku)
#if defined _perkmod_blind_luck_included
 #endinput
#endif
#define _perkmod_blind_luck_included

int g_iPerkBL;

void BlindLuck_Register()
{
	g_iPerkBL = Perks_Register("blind_luck", TREE_BOOMER);
	Events_Subscribe(PME_VOMIT, BL_OnIt);
	Events_MapStart_Add(BL_Reset);
	CreateTimer(1.0, BL_HudCheck, 0, TIMER_REPEAT|TIMER_FLAG_NO_MAPCHANGE);
}

void BL_Reset()
{
	for (int i = 1; i <= MaxClients; i++)
		if (IsValidClient(i)) SetEntProp(i, Prop_Send, "m_iHideHUD", 0);
}

void BL_OnIt(int victim, int attacker)
{
	if (!Perk_HasInfected(attacker, g_iPerkBL)) return;
	if (victim < 1 || victim > MaxClients) return;
	if (IsFakeClient(victim)) return;   // оригинал: ботов не слепим
	SetEntProp(victim, Prop_Send, "m_iHideHUD", 64);
}

Action BL_HudCheck(Handle timer)
{
	if (!g_hCvarEnable.BoolValue) return Plugin_Continue;
	for (int c = 1; c <= MaxClients; c++)
	{
		if (IsValidClient(c) && !IsFakeClient(c) && GetClientTeam(c) == 2
			&& GetEntProp(c, Prop_Send, "m_iHideHUD") == 64 && !Util_IsSlimed(c))
			SetEntProp(c, Prop_Send, "m_iHideHUD", 0);

		if (!IsValidClient(c) || GetClientTeam(c) != 3) continue;
		if (GetZombieClass(c) != 2) continue;   // boomer
		if (!Perk_HasInfected(c, g_iPerkBL)) continue;

		float mult = PerkCvar_Float(g_iPerkBL, "_cdmult", 0.5);   // кламп 0.01..1.0
		if (mult < 0.01) mult = 0.01;
		if (mult > 1.0)  mult = 1.0;

		int ab = GetEntPropEnt(c, Prop_Send, "m_customAbility");
		if (ab <= 0) continue;
		Ability_CompressCooldown(ab, mult);
	}
	return Plugin_Continue;
}
