// perkmod/ghost_rider.inc — почти невидимый джокей на езде (jockey №2)
#if defined _perkmod_ghost_rider_included
 #endinput
#endif
#define _perkmod_ghost_rider_included

int g_iPerkGR;

void GhostRider_Register()
{
	g_iPerkGR = Perks_Register("ghost_rider", TREE_JOCKEY);
	Events_Subscribe(PME_JOCKEY_RIDE,     GR_Ride);
	Events_Subscribe(PME_JOCKEY_RIDE_END, GR_RideEnd);
	Events_Subscribe(PME_DEATH,           GR_Death);
	Events_MapStart_Add(GR_Reset);
}

void GR_Reset()
{
	for (int i = 1; i <= MaxClients; i++)
		if (IsValidClient(i)) GR_SetAlpha(i, 255);
}

void GR_SetAlpha(int client, int alpha)
{
	if (!IsValidClient(client)) return;
	SetEntityRenderMode(client, RENDER_TRANSCOLOR);
	SetEntityRenderColor(client, 255, 255, 255, alpha);
}

void GR_Ride(int jockey, int victim)
{
	if (!Perk_HasInfected(jockey, g_iPerkGR)) return;
	int alpha = PerkCvar_Int(g_iPerkGR, "_alpha", 25);   // кламп 0..255
	if (alpha < 0)   alpha = 0;
	if (alpha > 255) alpha = 255;
	if (alpha >= 255) return;
	GR_SetAlpha(jockey, alpha);
}

void GR_RideEnd(int jockey, int victim)
{
	if (!IsValidClient(jockey) || GetClientTeam(jockey) != 3) return;
	if (GetZombieClass(jockey) != 5) return;
	GR_SetAlpha(jockey, 255);
}

void GR_Death(int victim, int attacker)
{
	if (!IsValidClient(victim) || GetClientTeam(victim) != 3) return;
	if (GetZombieClass(victim) != 5) return;
	GR_SetAlpha(victim, 255);
}
