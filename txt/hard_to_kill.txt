// perkmod/hard_to_kill.inc — усиленное инкап-здоровье (sur3 №3, дефолт 0.5, кламп 0.01..3.0)
#if defined _perkmod_hard_to_kill_included
 #endinput
#endif
#define _perkmod_hard_to_kill_included

int g_iPerkHK;

void HardToKill_Register()
{
	g_iPerkHK = Perks_Register("hard_to_kill", TREE_SUR3);
	Events_Subscribe(PME_INCAP, HK_Incap);
}

void HK_Incap(int client, int extra)
{
	if (!Perk_HasSurvivor(client, g_iPerkHK)) return;
	if (!IsValidClient(client) || GetClientTeam(client) != 2) return;
	DataPack dp;
	CreateDataTimer(0.5, HK_Delayed, dp, TIMER_FLAG_NO_MAPCHANGE);
	dp.WriteCell(GetClientUserId(client));
}

Action HK_Delayed(Handle timer, DataPack dp)
{
	dp.Reset();
	int c = GetClientOfUserId(dp.ReadCell());
	if (!c || !IsValidClient(c) || GetClientTeam(c) != 2) return Plugin_Stop;
	if (!Perk_HasSurvivor(c, g_iPerkHK)) return Plugin_Stop;

	float mult = PerkCvar_Float(g_iPerkHK, "_hpmult", 0.5);
	if (mult < 0.01) mult = 0.01;
	if (mult > 3.0)  mult = 3.0;

	// база — из конвара игры, не хардкод; ledge — отдельная база
	int base = (Util_IsOnLedge(c))
		? FindConVar("survivor_ledge_grab_health").IntValue
		: FindConVar("survivor_incap_health").IntValue;

	int add = RoundToNearest(float(base) * mult);
	SetEntPropFloat(c, Prop_Send, "m_healthBuffer", float(base + add));
	SetEntPropFloat(c, Prop_Send, "m_healthBufferTime", GetGameTime());
	return Plugin_Stop;
}
