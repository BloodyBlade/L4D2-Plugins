// perkmod/scattering_ram.inc — сила заряда + бонус HP (charger №1, дефолты 1.3 / 0.3)
#if defined _perkmod_scattering_ram_included
 #endinput
#endif
#define _perkmod_scattering_ram_included

int g_iPerkSR;

void ScatteringRam_Register()
{
	g_iPerkSR = Perks_Register("scattering_ram", TREE_CHARGER);
	Events_Subscribe(PME_CHARGE_IMPACT, SR_Impact);
	Events_Subscribe(PME_SPAWN,         SR_Spawn);
}

void SR_Spawn(int client, int extra)
{
	if (!Perk_HasInfected(client, g_iPerkSR)) return;
	if (GetZombieClass(client) != 6) return;
	CreateTimer(0.1, SR_ChangeHP, GetClientUserId(client), TIMER_FLAG_NO_MAPCHANGE);
}

Action SR_ChangeHP(Handle timer, int userid)
{
	int c = GetClientOfUserId(userid);
	if (c > 0 && IsValidClient(c) && Perk_HasInfected(c, g_iPerkSR) && GetZombieClass(c) == 6)
	{
		float mult = PerkCvar_Float(g_iPerkSR, "_hp", 0.3);   // кламп 0.0..3.0
		if (mult < 0.0) mult = 0.0;
		if (mult > 3.0) mult = 3.0;
		int base = GetEntProp(c, Prop_Data, "m_iMaxHealth");
		int add = RoundToCeil(float(base) * mult);
		SetEntProp(c, Prop_Data, "m_iMaxHealth", base + add);
		SetEntityHealth(c, GetClientHealth(c) + add);
	}
	return Plugin_Stop;
}

void SR_Impact(int attacker, int victim)
{
	if (!Perk_HasInfected(attacker, g_iPerkSR)) return;
	if (victim < 1 || victim > MaxClients || !IsValidClient(victim)) return;
	float force = PerkCvar_Float(g_iPerkSR, "_force", 1.3);   // кламп 1.0..3.0
	if (force <= 1.0) return;
	CreateTimer(0.1, SR_Force, GetClientUserId(victim), TIMER_FLAG_NO_MAPCHANGE);
}

Action SR_Force(Handle timer, int userid)
{
	int v = GetClientOfUserId(userid);
	if (!v || !IsValidClient(v)) return Plugin_Stop;
	float vel[3];
	GetEntPropVector(v, Prop_Data, "m_vecVelocity", vel);
	float force = PerkCvar_Float(g_iPerkSR, "_force", 1.3);
	vel[0] *= force; vel[1] *= force; vel[2] *= force;
	TeleportEntity(v, NULL_VECTOR, NULL_VECTOR, vel);
	return Plugin_Stop;
}
