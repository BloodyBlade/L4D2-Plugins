 #endinput

#define _perkmod_sleight_of_hand_included

int g_iPerkSoH;
int g_iSoHScaledRef[MAXPLAYERS+1];

void SleightOfHand_Register()
{
	g_iPerkSoH = Perks_Register("sleight_of_hand", TREE_SUR1);
	HookEvent("weapon_reload", SoH_Reload, EventHookMode_Post);
	Events_MapStart_Add(SoH_Reset);
}

void SoH_Reset()
{
	for (int i = 0; i <= MaxClients; i++) g_iSoHScaledRef[i] = 0;
}

void SoH_Reload(Event e, const char[] n, bool db)
{
	int c = GetClientOfUserId(e.GetInt("userid"));
	if (!Perk_HasSurvivor(c, g_iPerkSoH)) return;
	int wep = GetEntPropEnt(c, Prop_Send, "m_hActiveWeapon");
	if (wep <= 0) return;
	char cls[40];
	GetEntityClassname(wep, cls, sizeof(cls));
	if (StrContains(cls, "shotgun") == -1) return;

	float rate = PerkCvar_Float(g_iPerkSoH, "_rate", 0.5714);   // кламп 0.2..0.9
	if (rate < 0.2) rate = 0.2;
	if (rate > 1.0) rate = 1.0;
	if (g_iSoHScaledRef[c] == EntIndexToEntRef(wep)) return;    // гвард от компаунда
	g_iSoHScaledRef[c] = EntIndexToEntRef(wep);
	Util_ScaleShotgunReload(wep, cls, rate);
}
