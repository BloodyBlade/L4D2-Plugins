// perkmod/helping_hand.inc — Рука помощи (sur3)
// «Вы поднимаете других быстрее и даёте им бонусное здоровье.»
//
// Осознанное отклонение от оригинала PerkMod (README, п. 6): оригинал гнал глобальный
// конвар survivor_revive_duration; у нас конвар не трогается вовсе — конфликтов с чужими
// плагинами нет by design.
//
// Механика: бар ревайва в движке — визуал; реальное завершение ревайва движок считает
// по своему внутреннему таймеру. Поэтому:
//   1) на revive_begin переписываем длительность бара (визуал жертвы и ревайвера);
//   2) таймер по истечении сокращённого времени сам завершает ревайв —
//      EndBar-паттерн [VIP] SelfRevive by BloodyBlade: сброс бара/owner/target,
//      L4D_ReviveSurvivor, выдача буфера напрямую;
//   3) анти-дабл: если натива выбросит revive_success, ивент-хендлер буфер повторно не даст.
#if defined _perkmod_hh_included
 #endinput
#endif
#define _perkmod_hh_included

#define HH_BUFFER_COOP 15
#define HH_BUFFER_VS   10

bool g_bHHPending[MAXPLAYERS + 1];     // таймер завершения активен
bool g_bHHPluginDone[MAXPLAYERS + 1];  // ревайв завершён нами (анти-дабл буфера)

void HH_Init()
{
	HookEvent("revive_begin",   HH_EvReviveBegin);
	HookEvent("revive_end",     HH_EvReviveEnd);
	HookEvent("revive_success", HH_EvReviveSuccess);
}

void HH_OnMapStart()
{
	for (int i = 1; i <= MaxClients; i++)
	{
		g_bHHPending[i] = false;
		g_bHHPluginDone[i] = false;
	}
}

// сброс состояния: вызывать из ядра на player_spawn / player_death / round end
void HH_ResetClient(int client)
{
	g_bHHPending[client] = false;
	g_bHHPluginDone[client] = false;
}

static bool HH_IsSurvivor(int client)
{
	return IsValidClient(client) && GetClientTeam(client) == 2 && IsPlayerAlive(client);
}

// --- начало ревайва: ускоряем, если перк у ревайвера ---
void HH_EvReviveBegin(Event event, const char[] name, bool dontBroadcast)
{
	int reviver = GetClientOfUserId(event.GetInt("userid"));
	int victim  = GetClientOfUserId(event.GetInt("subject"));

	if (!HH_IsSurvivor(reviver) || !HH_IsSurvivor(victim) || reviver == victim) return;
	if (!Perk_HasSurvivor(reviver, g_iIdHelpingHand)) return;

	// база — фактическая длительность бара, которую движок уже выставил
	// (в ней учтён адреналин и прочие модификаторы); fallback на конвар, он гарантирован
	float base = Util_BarGetDuration(victim);
	if (base <= 0.0) base = FindConVar("survivor_revive_duration").FloatValue;

	float mult = PerkCvar_Float(g_iIdHelpingHand, "_timemult", 0.6);
	float dur = base * mult;
	if (dur <= 0.0) return;

	// сохранение набранного прогресса: если крюк сработал позже старта ревайва,
	// пересчитываем метку так, чтобы полузаполненный бар не отскакивал в ноль
	float elapsed = GetGameTime() - Util_BarGetStart(victim);
	float frac = (base > 0.0 && elapsed > 0.0) ? elapsed / base : 0.0;
	if (frac < 0.0) frac = 0.0;
	if (frac > 1.0) frac = 1.0;
	float start = GetGameTime() - frac * dur;

	Util_BarSetAt(victim, start, dur);
	Util_BarSetAt(reviver, start, dur);   // бар у обоих совпадает — иначе рассинхрон визуала

	g_bHHPending[victim] = true;
	g_bHHPluginDone[victim] = false;

	DataPack dp;
	CreateDataTimer(dur * (1.0 - frac), HH_Complete, dp);
	dp.WriteCell(victim);
	dp.WriteCell(reviver);
}

// --- досрочное завершение: наш таймер ---
Action HH_Complete(Handle timer, DataPack dp)
{
	dp.Reset();
	int victim  = dp.ReadCell();
	int reviver = dp.ReadCell();

	if (!g_bHHPending[victim]) return Plugin_Stop;   // уже прерван или завершён штатно
	g_bHHPending[victim] = false;

	if (!HH_IsSurvivor(victim)) return Plugin_Stop;

	// ревайв уже не идёт: прерван (revive_end) или движок успел сам
	if (Util_GetReviveOwner(victim) == -1) return Plugin_Stop;
	if (!Util_IsIncapacitated(victim) && !Util_IsOnLedge(victim)) return Plugin_Stop;

	// завершаем подъём принудительно
	g_bHHPluginDone[victim] = true;
	Util_BarReset(victim);
	Util_BarReset(reviver);
	Util_SetReviveOwner(victim, -1);
	Util_SetReviveTarget(reviver, -1);

	L4D_ReviveSurvivor(victim);
	HH_GiveBuffers(victim, reviver);
	return Plugin_Stop;
}

// --- штатное завершение ревайва (движок успел по своему таймеру или наша натива) ---
void HH_EvReviveSuccess(Event event, const char[] name, bool dontBroadcast)
{
	int victim = GetClientOfUserId(event.GetInt("subject"));
	int helper = GetClientOfUserId(event.GetInt("userid"));
	if (!HH_IsSurvivor(victim)) return;

	if (g_bHHPluginDone[victim])
	{
		// завершили мы — буфер уже выдан в HH_Complete, не дублируем
		g_bHHPluginDone[victim] = false;
		return;
	}

	// штатное завершение: снимаем флаг таймера, чтобы он отработал вхолостую
	g_bHHPending[victim] = false;

	if (!HH_IsSurvivor(helper) || !Perk_HasSurvivor(helper, g_iIdHelpingHand)) return;
	HH_GiveBuffers(victim, helper);
}

// --- ревайв прерван движком (урон/отход ревайвера) ---
void HH_EvReviveEnd(Event event, const char[] name, bool dontBroadcast)
{
	int victim = GetClientOfUserId(event.GetInt("subject"));
	if (!HH_IsSurvivor(victim)) return;

	if (g_bHHPluginDone[victim])
	{
		// это revive_end после нашего принудительного подъёма — всё уже чисто
		g_bHHPluginDone[victim] = false;
		return;
	}

	if (!g_bHHPending[victim]) return;   // перк не участвовал в этом ревайве
	g_bHHPending[victim] = false;

	// гасим бары и связку owner/target
	Util_BarReset(victim);
	int reviver = Util_GetReviveOwner(victim);
	if (reviver > 0 && HH_IsSurvivor(reviver))
	{
		Util_BarReset(reviver);
		Util_SetReviveTarget(reviver, -1);
	}
	Util_SetReviveOwner(victim, -1);
}

// --- буферы: жертве полный, ревайверу половина (поведение оригинала с 2.1.2) ---
void HH_GiveBuffers(int victim, int reviver)
{
	int buffer = Util_GameModeIsVersusOrScavenge()
		? PerkCvar_Int(g_iIdHelpingHand, "_buffer_vs", HH_BUFFER_VS)
		: PerkCvar_Int(g_iIdHelpingHand, "_buffer",    HH_BUFFER_COOP);

	Util_GiveTempBuffer(victim, buffer);
	if (HH_IsSurvivor(reviver) && reviver != victim)
		Util_GiveTempBuffer(reviver, buffer / 2);
}
