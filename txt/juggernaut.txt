// perkmod/juggernaut.inc — увеличенный HP танка (tank №1)
#if defined _perkmod_juggernaut_included
 #endinput
#endif
#define _perkmod_juggernaut_included

int g_iPerkJG;

void Juggernaut_Register()
{
	g_iPerkJG = Perks_Register("juggernaut", TREE_TANK);
	Events_Subscribe(PME_SPAWNTANK, JG_Spawn);
	Events_Subscribe(PME_SPAWN,     JG_Spawn);
}

void JG_Spawn(int client, int extra)
{
	if (!Perk_HasInfected(client, g_iPerkJG)) return;
	if (GetZombieClass(client) != 8) return;
	CreateTimer(0.1, JG_ChangeHP, GetClientUserId(client), TIMER_FLAG_NO_MAPCHANGE);
}

Action JG_ChangeHP(Handle timer, int userid)
{
	int c = GetClientOfUserId(userid);
	if (!c || !IsValidClient(c) || GetClientTeam(c) != 3 || GetZombieClass(c) != 8) return Plugin_Stop;
	if (!Perk_HasInfected(c, g_iPerkJG)) return Plugin_Stop;

	float mult = PerkCvar_Float(g_iPerkJG, "_hp", 0.5);   // кламп сверить
	if (mult < 0.0) mult = 0.0;
	if (mult > 3.0) mult = 3.0;

	// база из m_iMaxHealth: отражает z_tank_health и чужие правки, без хардкода
	int base = GetEntProp(c, Prop_Data, "m_iMaxHealth");
	int add = RoundToCeil(float(base) * mult);
	SetEntProp(c, Prop_Data, "m_iMaxHealth", base + add);
	SetEntityHealth(c, GetClientHealth(c) + add);
	return Plugin_Stop;
}
