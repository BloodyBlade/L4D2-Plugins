// perkmod/adrenal_glands.inc — кулдаун способностей танка режется (tank №6)
#if defined _perkmod_adrenal_glands_included
 #endinput
#endif
#define _perkmod_adrenal_glands_included

int g_iPerkAG;

void AdrenalGlands_Register()
{
	g_iPerkAG = Perks_Register("adrenal_glands", TREE_TANK);
	// тот же механизм компрессии кулдауна, что и у blind_luck (m_nextActivationTimer)
	CreateTimer(1.0, AG_Tick, 0, TIMER_REPEAT|TIMER_FLAG_NO_MAPCHANGE);
}

Action AG_Tick(Handle timer)
{
	if (!g_hCvarEnable.BoolValue) return Plugin_Continue;
	for (int c = 1; c <= MaxClients; c++)
	{
		if (!IsValidClient(c) || GetClientTeam(c) != 3) continue;
		if (GetZombieClass(c) != 8) continue;   // tank
		if (!Perk_HasInfected(c, g_iPerkAG)) continue;

		float mult = PerkCvar_Float(g_iPerkAG, "_cdmult", 0.5);   // кламп 0.01..1.0
		if (mult < 0.01) mult = 0.01;
		if (mult > 1.0)  mult = 1.0;

		int ab = GetEntPropEnt(c, Prop_Send, "m_customAbility");
		if (ab <= 0) continue;
		Ability_CompressCooldown(ab, mult);
	}
	return Plugin_Continue;
}
