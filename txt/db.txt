 #endinput

#define _perkmod_db_included

Database g_hDB;

void DB_Connect()
{
	char err[128];
	g_hDB = SQL_Connect("perkmod", false, err, sizeof(err));
	if (g_hDB == null) g_hDB = SQLite_UseDatabase("perkmod", err, sizeof(err));
	if (g_hDB != null)
	{
		char q[256];
		g_hDB.Format(q, sizeof(q), "CREATE TABLE IF NOT EXISTS perkmod (steamid TEXT PRIMARY KEY, s1 INTEGER, s2 INTEGER, s3 INTEGER, perk INTEGER, updated INTEGER)");
		g_hDB.Query(DB_Ignored, q);
		for (int i = 1; i <= MaxClients; i++)
			if (IsValidClient(i)) DB_LoadClient(i);
	}
}

void DB_Ignored(Database db, DBResultSet rs, const char[] err, any data) {}

void DB_SaveClient(int client)
{
	if (g_hDB == null || !IsValidClient(client) || IsFakeClient(client)) return;
	char sid[32];
	GetClientAuthId(client, AuthId_Steam2, sid, sizeof(sid));
	char q[256];
	g_hDB.Format(q, sizeof(q), "INSERT INTO perkmod (steamid, s1, s2, s3, perk, updated) VALUES ('%s', %d, %d, %d, %d, %d) ON CONFLICT(steamid) DO UPDATE SET s1=%d, s2=%d, s3=%d, perk=%d, updated=%d",
		sid, g_iSlot1[client], g_iSlot2[client], g_iSlot3[client], g_iInfectedPerk[client], GetTime(),
		g_iSlot1[client], g_iSlot2[client], g_iSlot3[client], g_iInfectedPerk[client], GetTime());
	g_hDB.Query(DB_Ignored, q);
}

void DB_LoadClient(int client)
{
	if (g_hDB == null || !IsValidClient(client) || IsFakeClient(client)) return;
	char sid[32];
	GetClientAuthId(client, AuthId_Steam2, sid, sizeof(sid));
	char q[256];
	g_hDB.Format(q, sizeof(q), "SELECT s1, s2, s3, perk FROM perkmod WHERE steamid = '%s'", sid);
	DataPack dp = new DataPack();
	dp.WriteCell(GetClientUserId(client));
	g_hDB.Query(DB_Loaded, q, dp);
}

void DB_Loaded(Database db, DBResultSet rs, const char[] err, any data)
{
	DataPack dp = view_as<DataPack>(data);
	dp.Reset();
	int client = GetClientOfUserId(dp.ReadCell());
	delete dp;
	if (!client || rs == null || !rs.FetchRow()) return;
	g_iSlot1[client] = rs.FetchInt(0);
	g_iSlot2[client] = rs.FetchInt(1);
	g_iSlot3[client] = rs.FetchInt(2);
	g_iInfectedPerk[client] = rs.FetchInt(3);
}
