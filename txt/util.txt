bool IsValidClient(int client)
{
	return client > 0 && client <= MaxClients && IsClientInGame(client);
}

int GetZombieClass(int client)
{
	return GetEntProp(client, Prop_Send, "m_zombieClass");
}

// --- PerkCvar: ленивое создание перковых конваров (perkmod_<perk><suffix>) ---
ConVar PerkCvar_Ref(int id, const char[] suffix, const char[] def)
{
	char name[64];
	Format(name, sizeof(name), "perkmod_%s%s", g_sPerkNames[id], suffix);
	ConVar h = FindConVar(name);
	if (h == null)
	{
		h = CreateConVar(name, def, "PerkMod: auto cvar", FCVAR_NOTIFY);
	}
	return h;
}
float PerkCvar_Float(int id, const char[] suffix, float def)
{
	char buf[16];
	FloatToString(def, buf, sizeof(buf));
	return PerkCvar_Ref(id, suffix, buf).FloatValue;
}
int PerkCvar_Int(int id, const char[] suffix, int def)
{
	char buf[16];
	IntToString(def, buf, sizeof(buf));
	return PerkCvar_Ref(id, suffix, buf).IntValue;
}

// --- рента скорости: m_flLaggedMovementValue под управлением владельца ---
float Util_GetBaseSpeed(int client)
{
	if (g_flSpeedBase[client] <= 0.0)
		g_flSpeedBase[client] = GetEntPropFloat(client, Prop_Data, "m_flLaggedMovementValue");
	return g_flSpeedBase[client];
}

void Util_SaveBaseSpeed(int client, int owner)
{
	if (g_iSpeedOwner[client] != SPEED_NONE && g_iSpeedOwner[client] != owner) return;
	g_iSpeedOwner[client] = owner;
	g_flSpeedBase[client] = GetEntPropFloat(client, Prop_Data, "m_flLaggedMovementValue") /
		Util_GetOwnerMult(client);   // снять чужой множитель, если был
}

float Util_GetOwnerMult(int client)
{
	switch (g_iSpeedOwner[client])
	{
		case SPEED_MB:  return PerkCvar_Float(g_iPerkMetabolicBoost(), "_mult", 1.4);
		case SPEED_SB:  return PerkCvar_Float(g_iPerkSpeedingBullet(), "_rate", 1.5);
		case SPEED_SD:  return PerkCvar_Float(g_iPerkSpeedDemon(), "_rate", 1.15);
		case SPEED_DD:  return PerkCvar_Float(g_iPerkDragDrop(), "_rate", 1.2);
		case SPEED_ADH: return 0.7;
		case SPEED_MS:  return PerkCvar_Float(g_iPerkMotionSickness(), "_rate", 0.85);
		case SPEED_EC:  return PerkCvar_Float(g_iPerkExtCond(), "_rate", 1.0);
		case SPEED_WIND: return 1.0;
	}
	return 1.0;
}

void Util_SetSpeed(int client, float mult)
{
	if (!IsValidClient(client)) return;
	SetEntPropFloat(client, Prop_Data, "m_flLaggedMovementValue", Util_GetBaseSpeed(client) * mult);
}

void Util_ClearSpeed(int client)
{
	g_iSpeedOwner[client] = SPEED_NONE;
	g_flSpeedBase[client] = 0.0;
	SetEntPropFloat(client, Prop_Data, "m_flLaggedMovementValue", 1.0);
}

void Util_RestoreOwnerSpeed(int client)
{
	Util_ClearSpeed(client);
}

bool Util_GameModeIsVersusOrScavenge()
{
	char gm[32];
	FindConVar("mp_gamemode").GetString(gm, sizeof(gm));
	return strcmp(gm, "versus") == 0 || strcmp(gm, "scavenge") == 0;
}

// --- temp HP (health buffer) ---
void Util_GiveTempBuffer(int client, int amount)
{
	if (amount <= 0) return;
	SetEntPropFloat(client, Prop_Send, "m_healthBuffer", GetEntPropFloat(client, Prop_Send, "m_healthBuffer") + float(amount));
	SetEntPropFloat(client, Prop_Send, "m_healthBufferTime", GetGameTime());
}

// --- счётчик ч/б (адаптировано из [VIP] SelfRevive by BloodyBlade) ---
void Util_IncrementReviveCount(int client)
{
	int count = GetEntProp(client, Prop_Send, "m_currentReviveCount") + 1;
	SetEntProp(client, Prop_Send, "m_currentReviveCount", count);
	if (count >= 2)
	{
		SetEntProp(client, Prop_Send, "m_bIsOnThirdStrike", 1);
		SetEntProp(client, Prop_Send, "m_isGoingToDie", 1);
	}
}

// --- макс-бозапас по конварам сервера (pack_rat) ---
int Util_GetMaxAmmo(const char[] cls)
{
	if (StrContains(cls, "smg")     != -1) return FindConVar("ammo_smg_max").IntValue;
	if (StrContains(cls, "rifle")  != -1) return FindConVar("ammo_rifle_max").IntValue;
	if (StrContains(cls, "shotgun")!= -1) return FindConVar("ammo_shotgun_max").IntValue;
	if (StrContains(cls, "sniper") != -1) return FindConVar("ammo_sniperrifle_max").IntValue;
	return 0;
}

// --- выдача предметов без чит-команд ---
void GiveItem(int client, const char[] classname, bool equip)
{
	if (equip)
	{
		int wep = GivePlayerItem(client, classname);
		if (wep > 0) EquipPlayerWeapon(client, wep);
	}
}

// --- компрессия кулдауна способности (метод Doku, m_nextActivationTimer) ---
void Ability_CompressCooldown(int ability, float mult)
{
	if (mult >= 1.0) return;
	// netprop-дот-путь; если на прогоне не сработает — fallback в геймдату (README)
	float ts = GetEntPropFloat(ability, Prop_Send, "m_nextActivationTimer.m_timestamp");
	float now = GetGameTime();
	float remain = ts - now;
	if (remain <= 0.0) return;
	SetEntPropFloat(ability, Prop_Send, "m_nextActivationTimer.m_timestamp", now + remain * mult);
}

// --- сверка имени активной способности ---
bool Util_AbilityNameMatches(int client, const char[] name)
{
	int ab = GetEntPropEnt(client, Prop_Send, "m_customAbility");
	if (ab <= 0) return false;
	char cls[64];
	GetEntityClassname(ab, cls, sizeof(cls));
	return strcmp(cls, name) == 0;
}

void Util_QueuePendingTank(int client)
{
	g_bDtTankRequested = true;
	g_iDtTankClient = client;
}

void Util_PumpPending()
{
	if (g_bDtTankRequested)
	{
		g_bDtTankRequested = false;
		float pos[3];
		if (g_iDtTankClient > 0 && IsClientInGame(g_iDtTankClient) && IsPlayerAlive(g_iDtTankClient))
			GetClientAbsOrigin(g_iDtTankClient, pos);
		else
			GetClientAbsOrigin(GetRandomSurvivor(), pos);  // fallback, или свой вариант

		L4D2_SpawnSpecial(8, pos, NULL_VECTOR);
	}
}

// --- табличные базы перезарядки дробовиков (пропы оружия, константы оригинала) ---
void Util_ScaleShotgunReload(int wep, const char[] cls, float rate)
{
	float s, i, e;
	if (strcmp(cls, "weapon_autoshotgun") == 0)       { s = 0.666666; i = 0.4;   e = 0.675;    }
	else if (strcmp(cls, "weapon_shotgun_spas") == 0) { s = 0.5;      i = 0.375; e = 0.699999; }
	else                                              { s = 0.5;      i = 0.5;   e = 0.6;      } // pump/chrome
	SetEntPropFloat(wep, Prop_Send, "m_reloadStartDuration",  s * rate);
	SetEntPropFloat(wep, Prop_Send, "m_reloadInsertDuration", i * rate);
	SetEntPropFloat(wep, Prop_Send, "m_reloadEndDuration",    e * rate);
}
