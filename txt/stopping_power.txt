// perkmod/stopping_power.inc — бонус-урон пушек выжившего (sur1 №1, дефолт 0.2, кламп 0.05..1.0)
#if defined _perkmod_stopping_power_included
 #endinput
#endif
#define _perkmod_stopping_power_included

int g_iPerkStP;

void StoppingPower_Register()
{
	g_iPerkStP = Perks_Register("stopping_power", TREE_SUR1);
	Dmg_Subscribe(StP_Dmg);
}

void StP_Dmg(int victim, int attacker, float &dmg, int &dmgtype, int weapon)
{
	if (dmg <= 0.0) return;
	if (!Perk_HasSurvivor(attacker, g_iPerkStP)) return;
	if (victim > 0 && victim <= MaxClients && GetClientTeam(victim) == 2) return;
	float mult = PerkCvar_Float(g_iPerkStP, "_dmg", 0.2);
	if (mult < 0.05) mult = 0.05;
	if (mult > 1.0)  mult = 1.0;
	dmg += dmg * mult;
}
