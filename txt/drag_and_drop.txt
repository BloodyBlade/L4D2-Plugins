#if defined _perkmod_drag_and_drop_included
 #endinput
#endif
#define _perkmod_drag_and_drop_included

int g_iPerkDD;

void DragAndDrop_Register()
{
	g_iPerkDD = Perks_Register("drag_and_drop", TREE_SMOKER);
	CreateTimer(0.5, DD_Tick, 0, TIMER_REPEAT|TIMER_FLAG_NO_MAPCHANGE);
}

Action DD_Tick(Handle timer)
{
	if (!g_hCvarEnable.BoolValue) return Plugin_Continue;
	for (int c = 1; c <= MaxClients; c++)
	{
		if (!IsValidClient(c) || GetClientTeam(c) != 3 || GetZombieClass(c) != 1) continue;
		bool dragging = GetEntPropEnt(c, Prop_Send, "m_tongueVictim") > 0;
		if (dragging && Perk_HasInfected(c, g_iPerkDD) && g_iSpeedOwner[c] != SPEED_DD)
			Util_SaveBaseSpeed(c, SPEED_DD), Util_SetSpeed(c, PerkCvar_Float(g_iPerkDD, "_rate", 1.2));
		else if (!dragging && g_iSpeedOwner[c] == SPEED_DD)
			Util_RestoreOwnerSpeed(c);
	}
	return Plugin_Continue;
}
